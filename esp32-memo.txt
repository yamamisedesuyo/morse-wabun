int charToIdx(char c){
  if(c==' ') return 0;
  if(c=='/') return 1;
  if(c=='?') return 2;
  if(c=='=') return 3;
  if(c>='0' && c<='9') return 4 + (c-'0');
  if(c>='A' && c<='Z') return 14 + (c-'A');
  if(c>='a' && c<='z') return 14 + (c-'a');
  return 0;
}

bool waitMs(int ms){
  unsigned long s=millis();
  while(millis()-s < (unsigned)ms){
    if(checkStopTouch()) return true;
    delay(5);
  }
  return false;
}
bool sendDot(){

  c=toupper(c);
  if(c==' '){ if(waitMs(dotLen()*4)) return true; return false; }
  if(c=='='){ if(sendDash()) return true; if(sendDot()) return true; if(sendDot()) return true; if(sendDot()) return true; if(sendDash()) return true; if(waitMs(dotLen()*2)) return true; return false; }

void enterEdit(int idx){
  editing=true; editIdx=idx; editMsg=msgs[idx]; editPos=0;
  if(editMsg.length()>0) pickerIdx=charToIdx(editMsg[0]); else pickerIdx=0;
  insertMode=true;
  drawEditScreen();
}

void handleEditTouch(uint16_t tx, uint16_t ty){
  if(ty<22 && tx>=188){ editing=false; drawScreen(); return; }
  if(tx>=60 && tx<=180 && ty>=100 && ty<=138){ pickerIdx=(pickerIdx+39)%40; drawEditScreen(); return; }
  if(tx>=60 && tx<=180 && ty>=190 && ty<=228){ pickerIdx=(pickerIdx+1)%40; drawEditScreen(); return; }
  if(tx>=2 && tx<=58 && ty>=143 && ty<=185){ if(editPos>0) editPos--; if(editPos<(int)editMsg.length()) pickerIdx=charToIdx(editMsg[editPos]); drawEditScreen(); return; }
  if(tx>=182 && tx<=238 && ty>=143 && ty<=185){ if(editPos<(int)editMsg.length()) editPos++; if(editPos<(int)editMsg.length()) pickerIdx=charToIdx(editMsg[editPos]); drawEditScreen(); return; }
  if(ty>=232 && ty<=262 && tx>=2 && tx<=118){ insertMode=!insertMode; drawEditScreen(); return; }
  if(ty>=232 && ty<=262 && tx>=122 && tx<=238){
    if(editMsg.length()>0){
      if(editPos<(int)editMsg.length()){ editMsg.remove(editPos,1); }
      else { editMsg.remove(editPos-1,1); editPos--; if(editPos<0) editPos=0; }
    }
    drawEditScreen(); return;
  }
  if(ty>=268 && ty<=298 && tx>=2 && tx<=78){
    char c = (pickerIdx==0 ? ' ' : CHAR_ORDER[pickerIdx]);
    if(insertMode){
      editMsg = editMsg.substring(0,editPos) + String(c) + editMsg.substring(editPos);
      editPos++;
    } else {
      if(editPos<(int)editMsg.length()){ editMsg.setCharAt(editPos,c); editPos++; }
      else { editMsg += String(c); editPos++; }
    }
    drawEditScreen(); return;
  }
  if(ty>=268 && ty<=298 && tx>=82 && tx<=158){
    msgs[editIdx]=editMsg;
    prefs.putString(("m"+String(editIdx)).c_str(), editMsg);
    editing=false; drawScreen(); return;
  }
  if(ty>=268 && ty<=298 && tx>=162 && tx<=238){
    editMsg="";
    editPos=0;
    pickerIdx=0;
    drawEditScreen(); return;
  }
}

void setup(){
  Serial.begin(115200);
  pinMode(PIN_DOT,INPUT_PULLUP); pinMode(PIN_DASH,INPUT_PULLUP);
  pinMode(PIN_KEY,OUTPUT); digitalWrite(PIN_KEY,LOW);

  prefs.begin("keyer",false);
  wpm=prefs.getInt("wpm",20);
  int ver=prefs.getInt("ver",0);
  if(ver<2){
    msgs[0]="CQ CQ CQ DE JS6UWQ JCC4713 PSE K";
    msgs[1]="UR RST 599 5NN BK";
    msgs[2]="73 TU E E";

    prefs.putInt("ver",2);

  }

}

void loop(){
  uint16_t tx,ty;
  if(tft.getTouch(&tx,&ty) && millis()-lastTouch>300){
    lastTouch=millis();
    if(showingHelp){
      if(ty>=268 && ty<=298 && tx>=82 && tx<=158){
        showingHelp=false; drawScreen();
      }
      return;
    }
    if(editing){ handleEditTouch(tx,ty); return; }
    Serial.printf("TOUCH x=%d y=%d\n", tx, ty);
    tft.fillCircle(tx,ty,4,TFT_WHITE);
    int W=tft.width(), H=tft.height();
    if(ty<22){
      if(tx>=86 && tx<=112){ wpm=max(5,wpm-1); prefs.putInt("wpm",wpm); drawScreen(); }
      if(tx>=118 && tx<=144){ wpm=min(50,wpm+1); prefs.putInt("wpm",wpm); drawScreen(); }
      if(tx>=188 && tx<=238){ showingHelp=true; drawHelpScreen(); }
    } else {
      int y0=24; int bh=(H-y0-4)/4 - 2;
      for(int i=0;i<4;i++){
        int y=y0+i*(bh+2);
        if(ty>=y && ty<y+bh){
          if(tx>W-84){ sendMsg(i); }
          else {
            tft.setTextColor(TFT_YELLOW,TFT_BLACK); tft.setTextSize(1); tft.setCursor(2,H-10);
            tft.print("HOLD 1s FOR EDIT...");
            unsigned long start=millis();
            bool held=true;
            while(millis()-start<1000){
              uint16_t hx,hy;
              if(!tft.getTouch(&hx,&hy)){ held=false; break; }
              delay(10);
            }
            if(held){
              enterEdit(i);
              while(tft.getTouch(&tx,&ty)) delay(10);
            } else {
              drawScreen();
            }
            lastTouch=millis();
          }
          break;
        }
      }
    }
  }
  if(!isPlaying && !editing && !showingHelp){
    if(digitalRead(PIN_DOT)==LOW){ keyOn(); while(digitalRead(PIN_DOT)==LOW) delay(1); keyOff(); delay(dotLen()); }
    if(digitalRead(PIN_DASH)==LOW){ keyOn(); delay(dotLen()*3); keyOff(); delay(dotLen()); }
  }
}
